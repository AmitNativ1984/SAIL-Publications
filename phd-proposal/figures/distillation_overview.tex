\newcommand{\netglyph}[3]{\tikz[x=1mm, y=1mm, baseline=0pt]{
  \foreach \l/\n in {0/4,1/5,2/5,3/4} {\foreach \i in {1,...,\n} {\coordinate (m\l-\i) at (\l*5, {(\i-(\n+1)/2)*1.4});}}
  \foreach \l/\n/\m in {0/4/5,1/5/5,2/5/4} {\pgfmathtruncatemacro{\nl}{\l+1}
    \foreach \i in {1,...,\n} {\foreach \j in {1,...,\m} {\draw[#1, line width=0.2pt] (m\l-\i) -- (m\nl-\j);}}}
  \foreach \l/\n in {0/4,1/5,2/5,3/4} {\foreach \i in {1,...,\n} {\fill[#2] (m\l-\i) circle (0.5); \draw[#3, line width=0.35pt] (m\l-\i) circle (0.5);}}
}}
