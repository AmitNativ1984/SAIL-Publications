\newcommand{\mlpglyph}{\tikz[x=1mm, y=1mm, baseline=0pt]{
  \def\layers{{4,5,5,3}}
  \foreach \l/\n in {0/4,1/5,2/5,3/4} {\foreach \i in {1,...,\n} {\coordinate (m\l-\i) at (\l*6.5, {(\i-(\n+1)/2)*1.6});}}
  \foreach \l/\n/\m in {0/4/5,1/5/5,2/5/4} {\pgfmathtruncatemacro{\nl}{\l+1}
    \foreach \i in {1,...,\n} {\foreach \j in {1,...,\m} {\draw[sfBlue!55, line width=0.2pt] (m\l-\i) -- (m\nl-\j);}}}
  \foreach \l/\n in {0/4,1/5,2/5,3/4} {\foreach \i in {1,...,\n} {\fill[white] (m\l-\i) circle (0.55); \draw[sfBlue!90!black, line width=0.35pt] (m\l-\i) circle (0.55);}}
}}
